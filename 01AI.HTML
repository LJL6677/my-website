function submitConfusion(module) { const inputMap = { 'pre': 'preConfusionInput', 'inclass': 'inConfusionInput', 'quiz': 'quizConfusionInput' }; const studentIdMap = { 'pre': 'studentId', 'inclass': 'inStudentId', 'quiz': 'quizStudentId' }; const nameMap = { 'pre': 'studentName', 'inclass': 'inStudentName', 'quiz': 'quizStudentName' }; const input = document.getElementById(inputMap[module]); const studentIdEl = document.getElementById(studentIdMap[module]); const nameEl = document.getElementById(nameMap[module]); const text = input.value.trim(); const studentId = studentIdEl ? studentIdEl.value.trim() : ''; const studentName = nameEl ? nameEl.value.trim() : ''; const area = input.closest('.confusion-area'); const activeTags = area.querySelectorAll('.tag.active'); const selectedTypes = Array.from(activeTags).map(el => el.dataset.type || el.textContent.trim()); if (!text && selectedTypes.length === 0) { alert('请描述你的疑问或勾选困惑选项！'); return; } if (!studentId) { alert('请先填写学号！'); return; } if (!studentName) { alert('请先填写姓名！'); return; } // 👈 新增：没名字不许提交 const btn = area.querySelector('.btn-confusion'); btn.disabled = true; btn.textContent = '⏳ 提交中...'; try { // 1. 先把学生信息保存好（确保 studentProfiles 里有这个名字） saveStudentProfile(studentId, studentName, { allowNameOverwrite: true }); // 2. 更新本地 confusionData if (!confusionData[studentId]) confusionData[studentId] = []; const existingIdx = confusionData[studentId].findIndex(e => e.module === module); const entry = { module, text: text || '(勾选困惑)', selectedTypes, timestamp: new Date().toISOString() }; if (existingIdx >= 0) { confusionData[studentId][existingIdx] = entry; } else { confusionData[studentId].push(entry); } localStorage.setItem('confusion_data', JSON.stringify(confusionData)); // 3. 同步到云端（studentName 直接从输入框拿，确保正确） saveModuleData(studentId, 'confusion', null, { studentId, studentName, // 👈 直接用输入框的名字 entries: confusionData[studentId], updatedAt: new Date().toISOString() }).catch(e => console.warn('困惑同步云端失败：', e)); // 4. UI 反馈 if (input) { input.value = ''; input.placeholder = '✅ 已提交'; } activeTags.forEach(el => el.classList.remove('active')); btn.textContent = '✅ 已提交'; btn.style.background = '#48bb78'; setTimeout(() => { if (input) { input.placeholder = '还有其他困惑？请描述...'; } btn.textContent = '📤 提交疑问'; btn.style.background = '#d69e2e'; btn.disabled = false; }, 3000); } catch (error) { btn.textContent = '❌ 提交失败'; setTimeout(() => { btn.textContent = '📤 提交疑问'; btn.disabled = false; }, 2000); } }
